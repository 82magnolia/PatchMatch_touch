%% NOTE: the modality and scale blocks score the COARSE transfer, since that is the
%% stage those two choices act on. The refinement block scores the final prediction.
%% The 4x row of the scale block is the same run as the surface-normal row of the
%% modality block, and the "w/o neural refinement" row is that same run's output.
\begin{table}[t]
  \centering
  \small
  \caption{\textbf{Ablation study} on a 20-object subset of the full pipeline benchmark. The top
    block varies the rendered modality used for retrieval and for feature matching, with the field
    of view fixed at $4\times$ the sensor footprint. The middle block varies that field of view,
    with the modality fixed to surface normals. Both of these blocks score the coarse transfer,
    since that is the stage the two choices act on, so their numbers are not comparable to the
    bottom block, which removes components of the refinement stage and scores the final
    prediction. The default configuration is marked $^\dagger$ and is the same run in all three
    blocks. The row marked $^\ddagger$ uses a checkpoint that differs from the full model in its
    training schedule as well as in its conditioning, so it overstates the effect of normal map
    concatenation on its own.}
  \label{tab:ablation}
  \begin{tabular}{l l c c c}

      & Height map       & 26.61 & 0.8153 & 0.2853 \\

      & Surface normals$^\dagger$ & 27.71 & \textbf{0.8244} & \textbf{0.2797} \\
    \hline
    \multirow{3}{*}{\rotatebox{90}{\scriptsize Scale}}

      & $4\times^\dagger$ & 27.71 & \textbf{0.8244} & \textbf{0.2797} \\

    \multirow{4}{*}{\rotatebox{90}{\scriptsize Refinement}}
      & w/o neural refinement       & 27.71 & 0.8244 & 0.2797 \\
      & w/o temporal FiLM           & 29.90 & 0.9022 & 0.1735 \\
      & w/o normal concatenation$^\ddagger$ & 25.38 & 0.8540 & 0.2722 \\
